Executable Verifier Source Code (repro/trackio_runner.py)

def verify_wang2024starnet() -> dict[str, dict[str, Any]]:
    from spd_learn.models import STaRNet

    X_tr, y_tr, X_te, y_te = make_synthetic_eeg(n_trials=80, n_chans=6, n_times=96)
    model = STaRNet(
        n_chans=6,
        n_outputs=2,
        n_patches=4,
        temporal_filters=8,
        spatial_filters=8,
        bimap_dim=6,
    )

    # Claim 1 & 2: Structural & Signal (Both Riemannian and SE branches receive non-zero gradients)
    out = model(X_tr[:4])
    out.sum().backward()
    riem_grad = float(
        sum(p.grad.norm().item() for p in model.bimap.parameters() if p.grad is not None)
    )
    eucl_grad = float(
        sum(
            p.grad.abs().sum().item()
            for p in model.euclid_branch.parameters()
            if p.grad is not None
        )
    )

    # Claim 3: Analytical
    res = train_and_eval_classifier(model, X_tr, y_tr, X_te, y_te, epochs=35)

    return {
        "wang2024starnet_c1": {
            "output_shape": list(out.shape),
            "passed": list(out.shape) == [4, 2],
            "scope": "full",
        },
        "wang2024starnet_c2": {
            "riemannian_bimap_grad_norm": riem_grad,
            "euclidean_se_grad_sum": eucl_grad,
            "passed": riem_grad > 0.0 and eucl_grad > 0.0,
            "scope": "full",
        },
        "wang2024starnet_c3": {
            "test_acc": res["test_acc"],
            "passed": res["test_acc"] >= 0.75,
            "scope": "toy",
        },
    }

Paper Abstract

STaRNet combines a Riemannian patch-covariance branch with a parallel Euclidean spatio-temporal squeeze-and-excitation branch, fusing geometric spatial covariance features with multi-scale temporal dynamics for motor imagery decoding.
